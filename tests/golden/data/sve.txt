### method c_simd_size_function()
static inline size_t get_simd_size() {
    size_t byte_size = 0;
    __asm__ volatile(
        "cntb %[byte_size]\n\t"
    : [byte_size] "=r" (byte_size)
    :
    :
    );
    return byte_size;
}
