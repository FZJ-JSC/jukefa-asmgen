static inline size_t get_simd_size() {
    size_t byte_size = 0;
    __asm__ volatile(
        "th.vsetvli %[byte_size], zero, e8, m1\n\t"
    : [byte_size] "=r" (byte_size)
    :
    :
    );
    return byte_size;
}
